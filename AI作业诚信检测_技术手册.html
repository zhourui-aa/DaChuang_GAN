0　手册定位与使用说明

  • 面向：团队全体（数据与样本采集岗 / Java 业务微服务岗 / 检测编排岗 / AI 检测岗 / 前端证据可视化岗 / 申诉流程岗 / 队长兼 PM）。
  • 给什么：每章给关键骨架（建表 DDL / 接口签名 / 检测特征 DSL / LangGraph graph / MCP 工具契约 / RocketMQ Topic / ES mapping / 检测评测口径 / yaml）。完整 CRUD 复用课程实训模板（计算机轨「智学云」单体后台模板 + 「医枢云」微服务工程模板），不重复造轮子。
  • 怎么用：立项期冻结数据模型 v1（§3）与检测特征定义 DSL v1（§3.4）；按章节顺序开发，前端与后端并行，检测算法与评测回归在中后段接入。
  • 配套：《落地执行手册》管排期 / 分工 / 赛事对位 / 答辩。
  • 技术合规基线（贯穿全册，不可让步）：
    • 本地化：检测全程在私有化本地模型（Ollama / vLLM）内完成，作业原文不出校园网、不上传第三方接口。
    • 只给证据不给定性：输出为特征值 + 证据片段 + 置信度 + 复核建议，不输出「该生作弊 / 系 AI 生成」结论。
    • 可溯源：每条证据必须能定位回原文的字符区间（detect_evidence.begin_pos / end_pos），不可溯源的特征不写入报告。
    • 最小必要 + 权限分级：只采集作业文本与必要的课程/提交元数据，不采集与检测无关的数据；任课教师 / 教研室 / 学院 / 管理员四级权限；学生姓名学号在列表页脱敏。
    • 申诉优先：任何高关注结果自动开启申诉通道入口，教师端不得只展示检测结论而无申诉路径。
    • 阈值可回看：检测配置版本化存储（detect_configs），每次检测结果记录所用配置版本，支持按版本回放。

2　Monorepo 工程骨架

四条架构决策（贯穿全册）

  1. 检测与业务解耦：Java 业务侧只管「任务、结果、证据、工单、权限」，检测算法全部在 Python 侧。算法迭代不动业务表结构，业务扩容不动算法代码，两侧只靠 §7 的协议契约耦合。
  2. 特征与证据分离存储：特征值（数值/评分）进 MySQL 明细表，全文与证据片段进 Elasticsearch，原文文件进 MinIO。报告组装时按 submission_id 关联三方，避免大文本塞进业务库。
  3. 检测可降级：AI 检测服务超时/熔断时，降级为「纯统计特征模式」（困惑度近似值 + 突发度 + 词汇丰富度 + 结构规律性），仍能给出特征证据，只是不含大模型语义分析。
  4. 结论可回归：每次检测记录所用模型版本与配置版本；评测集（留出集）跑回归，防止调阈值/换模型后效果回退。

3　核心数据模型（MySQL DDL）与检测特征 DSL

立项冻结 v1，全队签字，后续只加不改。共 14 张表，分四组：课程与提交（业务源）、检测（配置/任务/特征/证据/结果）、申诉（工单/复核）、治理（数据集/审计）。数据分级：L1 提交标识（学号脱敏展示）；L2 作业正文与检测特征（任课教师可见）；L3 证据明细与申诉记录（教研室/学院可见）；L4 申诉复核详情（仅学院级）。所有访问经 auth 服务审计留痕。

3.1 课程与提交

CREATE TABLE courses (
  course_id       BIGINT PRIMARY KEY AUTO_INCREMENT,
  course_code     VARCHAR(30) UNIQUE NOT NULL,
  course_name     VARCHAR(100),
  college_id      BIGINT NOT NULL,               -- 开课学院
  teacher_id      BIGINT NOT NULL,               -- 任课教师
  term            VARCHAR(10),                    -- 2026-2027-1
  enabled         TINYINT DEFAULT 1
);

CREATE TABLE classes (
  class_id        BIGINT PRIMARY KEY AUTO_INCREMENT,
  class_name      VARCHAR(60),
  college_id      BIGINT NOT NULL,
  major           VARCHAR(60),
  grade           SMALLINT
);

CREATE TABLE students (
  student_id      BIGINT PRIMARY KEY AUTO_INCREMENT,
  student_no      VARCHAR(20) UNIQUE NOT NULL,   -- 学号（列表页脱敏）
  name_mask       VARCHAR(40),                   -- 脱敏姓名（张*）
  class_id        BIGINT,
  INDEX idx_class(class_id)
);

CREATE TABLE assignments (
  assignment_id   BIGINT PRIMARY KEY AUTO_INCREMENT,
  course_id       BIGINT NOT NULL,
  title           VARCHAR(120),
  require_words   INT,                            -- 要求字数（短文本检测门槛）
  allow_appeal    TINYINT DEFAULT 1,
  deadline        DATETIME,
  created_at      DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE submissions (
  submission_id   BIGINT PRIMARY KEY AUTO_INCREMENT,
  assignment_id   BIGINT NOT NULL,
  student_id      BIGINT NOT NULL,
  file_key        VARCHAR(200),                   -- MinIO submission-files 对象键
  raw_text        LONGTEXT,                       -- 解析后纯文本（去噪、去格式）
  char_count      INT,                            -- 去噪后字符数
  text_sha256     CHAR(64),                       -- 文本指纹（重复检测 / 缓存去重）
  submit_status   VARCHAR(20) DEFAULT '已提交',   -- 已提交/解析失败/待检测
  created_at      DATETIME DEFAULT CURRENT_TIMESTAMP,
  UNIQUE KEY uk_assign_student(assignment_id, student_id),
  INDEX idx_text_sha(text_sha256)
);

3.3 特征、证据与结果

CREATE TABLE detect_features (
  feature_id      BIGINT PRIMARY KEY AUTO_INCREMENT,
  task_id         BIGINT NOT NULL,
  feature_code    VARCHAR(40) NOT NULL,           -- PERPLEXITY / PARAPHRASE_TRACE ...
  feature_value   DECIMAL(8,4),                   -- 原始指标值（如困惑度 12.83）
  feature_score   DECIMAL(5,4),                   -- 归一化风险分 0-1
  weight          DECIMAL(4,3),                   -- 本次生效权重（来自配置版本）
  feature_text    VARCHAR(300),                   -- 特征自然语言描述（可解释）
  source          VARCHAR(20),                    -- STATS（统计）/ LLM（模型语义）
  INDEX idx_task(task_id)
);

CREATE TABLE detect_evidence (
  evidence_id     BIGINT PRIMARY KEY AUTO_INCREMENT,
  task_id         BIGINT NOT NULL,
  feature_code    VARCHAR(40),
  begin_pos       INT NOT NULL,                   -- 证据片段在原文中的起始字符位
  end_pos         INT NOT NULL,                   -- 结束位（左闭右开）
  snippet         VARCHAR(500),                   -- 证据片段原文
  reason          VARCHAR(300),                   -- 何以成为证据（可解释）
  INDEX idx_task_code(task_id, feature_code)
);

CREATE TABLE detect_results (
  result_id       BIGINT PRIMARY KEY AUTO_INCREMENT,
  task_id         BIGINT NOT NULL UNIQUE,
  submission_id   BIGINT NOT NULL,
  risk_score      DECIMAL(5,4),                   -- 综合分 0-1（加权和）
  risk_level      VARCHAR(20),                    -- 低关注/中关注/高关注
  confidence      DECIMAL(5,4),                   -- 置信度（证据充分性）
  summary         TEXT,                           -- 检测说明（含"仅供参考"声明）
  review_suggestion VARCHAR(300),                 -- 人工复核建议
  config_version  VARCHAR(20),                    -- 所用配置版本（回看用）
  created_at      DATETIME DEFAULT CURRENT_TIMESTAMP
);

3.4 检测特征定义 DSL（JSON 示例，v1 冻结）

主特征 3 类（困惑度 / 改写痕迹 / 行文轨迹）+ 辅证 3 类（句长突发度 / 词汇丰富度 / 结构规律性）。每类给 code / method / direction / weight / score_map / tag_text_template；权重之和为 1。方向 inverse 表示指标越低越可疑（如困惑度），direct 表示越高越可疑（如结构规律性）。

// detect_configs.feature_dsl v1（节选 3 条主特征 + 3 条辅证）
{
  "version": "v1",
  "features": [
    {
      "code": "PERPLEXITY",
      "name": "文本困惑度",
      "method": "local_llm_logprob",          // 本地模型逐 token logprob 求平均负对数似然
      "direction": "inverse",
      "weight": 0.25,
      "score_map": { "<=8": 1.0, "8-15": 0.6, "15-30": 0.3, ">30": 0.0 },
      "tag_text_template": "全文平均困惑度 {value}，低于人工撰写常见区间，句式可预测性偏高"
    },
    {
      "code": "PARAPHRASE_TRACE",
      "name": "改写痕迹",
      "method": "llm_semantic_scan",           // 模型定位"同义替换密集但信息增益低"的段落
      "direction": "direct",
      "weight": 0.25,
      "score_map": { "<=0.2": 0.0, "0.2-0.4": 0.3, "0.4-0.6": 0.6, ">0.6": 1.0 },
      "tag_text_template": "检测到 {count} 处改写痕迹，表现为同义替换密集、句意重复度高"
    },
    {
      "code": "WRITING_TRAJECTORY",
      "name": "行文轨迹",
      "method": "llm_trajectory_analysis",      // 段落展开是否符合人工写作的"起伏—修正—回扣"轨迹
      "direction": "direct",
      "weight": 0.20,
      "score_map": { "<=0.2": 0.0, "0.2-0.5": 0.4, ">0.5": 0.8 },
      "tag_text_template": "行文轨迹过于平直，缺少人工写作常见的修改与回扣痕迹"
    },
    {
      "code": "BURSTINESS",
      "name": "句长突发度",
      "method": "stats_sentence_length_variance",
      "direction": "inverse",
      "weight": 0.12,
      "score_map": { "<=8": 0.8, "8-20": 0.4, ">20": 0.0 },
      "tag_text_template": "句长方差 {value}，句子长度分布过于均匀"
    },
    {
      "code": "LEXICAL_DIVERSITY",
      "name": "词汇丰富度",
      "method": "stats_type_token_ratio",
      "direction": "inverse",
      "weight": 0.10,
      "score_map": { "<=0.35": 0.7, "0.35-0.45": 0.3, ">0.45": 0.0 },
      "tag_text_template": "实词类符形符比 {value}，用词丰富度偏低"
    },
    {
      "code": "STRUCT_PATTERN",
      "name": "结构规律性",
      "method": "stats_paragraph_template_ratio",   // 段落开头句式 / 标点分布的模板化比例
      "direction": "direct",
      "weight": 0.08,
      "score_map": { "<=0.3": 0.0, "0.3-0.6": 0.4, ">0.6": 0.8 },
      "tag_text_template": "{value} 的段落呈模板化展开（开头句式高度一致）"
    }
  ],
  "aggregate": "weighted_sum",
  "min_chars": 200,
  "notes": "字符数 < min_chars 时不计综合分，只出特征明细，直接标注'短文本证据不足'"
}

合成规则：risk_score = Σ(feature_score × weight)；risk_level 由 detect_configs.risk_threshold 分档（低/中/高关注）。置信度与分数分开计算：可用特征数 ≥5 且每条主特征都有证据片段 → 高；3–4 条 → 中；<2 条或字符数不足 → 低，并在 summary 中显式标注「证据不足，仅供教师复核参考，不建议据此发起处理」。

6　AI 运行时落地（LangGraph 检测 graph + MCP Server）

AI 层职责单一：吃进一段作业文本，吐出「特征值 + 证据片段 + 置信度 + 复核建议」。不做定性判定，不做成绩评定，不外传文本。整条链路跑在本地模型上（Ollama / vLLM），模型不可用时降级为纯统计特征模式。

6.1 LangGraph graph 定义（Python）

from typing import TypedDict, List
from langgraph.graph import StateGraph, END

class DetectState(TypedDict):
    submission_id: int
    text: str                      # 去噪后纯文本
    char_count: int
    chunks: List[dict]             # 分句/分段结果（带字符区间）
    stats_features: List[dict]       # 统计特征：困惑度近似 / 突发度 / 词汇丰富度 / 结构规律性
    llm_features: List[dict]         # 模型语义特征：改写痕迹 / 行文轨迹
    evidences: List[dict]           # 证据片段 [{feature_code, begin_pos, end_pos, snippet, reason}]
    features: List[dict]            # 合并后的特征明细（含 score / weight / text）
    risk_score: float
    risk_level: str
    confidence: float
    summary: str
    suggestion: str
    case_refs: List[dict]           # RAG 召回的历史相似案例 / 校规条文
    guardrail_passed: bool

def preprocess(state: DetectState) -> DetectState:
    # 分句分段、去噪、字数校验；< min_chars 直接标记证据不足
    state["chunks"] = split_sentences(state["text"])
    state["char_count"] = len(state["text"])
    return state

def extract_stats(state: DetectState) -> DetectState:
    # 统计侧特征：句长突发度 / 类符形符比 / 段落模板化比例
    # 困惑度：本地模型逐 token logprob 求平均负对数似然（不产生文本输出）
    state["stats_features"] = stats_extractor.run(state["text"], state["chunks"])
    return state

def agent_analyze(state: DetectState) -> DetectState:
    # 语义侧特征：改写痕迹 + 行文轨迹，走本地大模型 + 结构化输出约束
    # 输出必须是 schema 约束的 JSON，禁止自由文本
    result = llm_agent.invoke(build_detect_prompt(state))   # 返回 {features:[...], evidences:[...]}
    state["llm_features"] = result["features"]
    state["evidences"] = result["evidences"]
    return state

def retrieve_cases(state: DetectState) -> DetectState:
    # RAG：从案例库 + 校规条文召回"同类写法"的复核结论与条文依据
    state["case_refs"] = case_rag.search(state["text"], top_k=3)
    return state

def aggregate(state: DetectState) -> DetectState:
    # 按 feature_dsl 权重合成 risk_score / risk_level，并单独算 confidence
    state["features"] = merge(state["stats_features"], state["llm_features"])
    state["risk_score"] = weighted_sum(state["features"])
    state["risk_level"] = to_level(state["risk_score"], cfg.risk_threshold)
    state["confidence"] = compute_confidence(state)
    return state

def guardrail_check(state: DetectState) -> DetectState:
    # 护栏：① 不出现"作弊/抄袭/AI 代写"等定性词；② 结论必须声明"仅供参考"；
    # ③ 每条证据必须能定位回原文区间；④ 置信度低时不得给高关注等级
    ok = guardrails.validate(state)
    if not ok:
        state["summary"] = build_neutral_summary(state)   # 降级为中性描述
        state["risk_level"] = "低关注"
    state["guardrail_passed"] = ok
    return state

graph = StateGraph(DetectState)
graph.add_node("preprocess", preprocess)
graph.add_node("extract_stats", extract_stats)
graph.add_node("agent_analyze", agent_analyze)
graph.add_node("retrieve_cases", retrieve_cases)
graph.add_node("aggregate", aggregate)
graph.add_node("guardrail", guardrail_check)
graph.set_entry_point("preprocess")
graph.add_edge("preprocess", "extract_stats")
graph.add_edge("extract_stats", "agent_analyze")
graph.add_edge("agent_analyze", "retrieve_cases")
graph.add_edge("retrieve_cases", "aggregate")
graph.add_edge("aggregate", "guardrail")
graph.add_edge("guardrail", END)
app = graph.compile()

降级分支：extract_stats 之后先探一次 detect:model:health；模型不可用时跳过 agent_analyze 与 retrieve_cases，直接进 aggregate，并把 degraded=1 回传 Java 侧，报告中显式标注「本次仅统计特征，未含模型语义分析」。

6.4 案例 RAG 工程（证据样例库 + 校规条文）

  • 数据源：① 校内既往已复核归档的争议案例（脱敏后的片段 + 复核结论，来自 appeal_reviews 与 evidence-samples）；② 课程考核办法、学籍管理规定、学术规范说明等条文。
  • 切片：案例按「问题段落 + 结论」成对切片，chunk_size 384，overlap 64；条文按条款切分。
  • 检索：BM25 + 向量混合（bge-small-zh），Rerank 用 bge-reranker，top_k=3。
  • 忠实度：生成建议中引用的条文必须能在原文定位，不可定位则删除该引用（避免"编造校规"）。
  • 不参与评分：案例召回结果只作参考展示，不进入 risk_score 计算——避免"因为像旧案例所以判高关注"的循环论证。

6.5 护栏（Guardrails）与 Agent Eval

维度规则违规处置
禁止定性词输出不得含「作弊 / 抄袭 / 代写 / 造假 / 认定」等词命中即改写为中性描述；仍命中则降级为特征原文
必须可溯源每条证据必须有 begin_pos/end_pos 且能在原文定位定位失败的证据直接丢弃，不写报告
仅供参考声明summary 必须含「本结果供教师复核参考，不作为处理依据」缺失则自动补写
置信度约束置信度低（可用特征 <2 条或字数不足）不得判「高关注」自动降级为「低关注」并提示证据不足
隐私不外传文本只送本地模型；调用链不含外部公网地址配置层硬编码白名单，越界请求直接拒绝
# eval/run_eval.py —— 留出集回归（Agent Eval）
# 指标口径：holdout 200 份（人工 100 / AI 100），逐份跑 detect_text 后统计
#   召回率 recall   = 判为"中/高关注"的 AI 文本数 / AI 文本总数     目标 ≥ 0.85
#   误报率 fpr      = 判为"中/高关注"的人工文本数 / 人工文本总数   目标 ≤ 0.10
#   证据命中率 cover = 有至少 1 条可定位证据的样本数 / 总样本数      目标 ≥ 0.95
#   护栏通过率 guard = 通过 guardrails 校验的输出数 / 总输出数       目标 = 1.00
python eval/run_eval.py --config v1 --holdout datasets/holdout_200.jsonl \
       --report eval/reports/v1.json

7　两栈对接（Java ↔ Python 协议契约）

Java 微服务（业务与数据服务化）与 Python AI 层（检测与解释）以 MCP 协议对接。Java 侧 integrity-detect 为 MCP Client，Python 侧 integrity-ai-detector 为 MCP Server。降级策略：AI 服务超时 / 熔断 / 探针失败时，detect 服务切纯统计特征模式，仍出特征明细与证据，只是不含语义分析。

调用方被调方协议超时降级
detect（Java）ai-detector（Python MCP Server）· detect_textMCP over HTTP30s（长文本）/ 15s（短文本）纯统计特征模式，degraded=1
detect（Java）ai-detector · explain_featureMCP over HTTP8s直接用 feature_text_template 模板渲染
appeal（Java）ai-detector · recommend_reviewMCP over HTTP8s返回固定建议文案「请结合课程要求与写作过程材料人工复核」
evidence（Java）ai-detector · search_similar_casesMCP over HTTP5s直接走 ES more_like_this
// Java 侧 MCP Client（detect）
@FeignClient(name = "ai-detector", url = "${ai.mcp.url}", fallback = AiDetectorFallback.class)
public interface AiDetectorClient {
    @PostMapping("/mcp/tools/detect_text")
    DetectOutputDTO detectText(@RequestBody DetectInput input);
}

// 降级兜底：纯统计特征（复用 Java 侧轻量统计，不依赖模型）
@Component
public class AiDetectorFallback implements AiDetectorClient {
    public DetectOutputDTO detectText(DetectInput input) {
        List<FeatureItem> f = statsOnlyAnalyzer.run(input.getText());  // 突发度/词汇丰富度/结构规律性
        return DetectOutputDTO.builder()
                .features(f).evidences(List.of())
                .confidence(0.3).degraded(true)
                .summary("AI 检测服务暂不可用，本次仅统计特征，未含语义分析，请勿据此处理。")
                .build();
    }
}

契约稳定性要求：detect_text / explain_feature / recommend_review / search_similar_cases 四个工具的出参字段一旦冻结（配置冻结节点内）不再增删，只允许在末尾追加可选字段；Java 侧 DTO 用 @JsonIgnoreProperties(ignoreUnknown = true) 保持前后兼容。

注：Spring AI / LangChain4j 可作为无 Python 运行环境时的纯 Java 等价方案（用本地模型 HTTP 接口 + Java 侧统计特征），但本项目以 LangGraph + MCP 为 AI 主方案，Spring AI 仅作降级备选，不作为唯一实现。

9.2 证据高亮渲染（核心交互）

// EvidenceHighlight.vue：把多段证据区间渲染到原文上，可点击跳转并联动说明卡
const segments = computed(() => {
  const txt = props.rawText;                 // 后端返回的 submissions.raw_text
  const evs = [...props.evidences].sort((a, b) => a.beginPos - b.beginPos);
  let cur = 0; const out = [];
  for (const e of evs) {
    if (e.beginPos > cur) out.push({ text: txt.slice(cur, e.beginPos), ev: null });
    out.push({ text: txt.slice(e.beginPos, e.endPos), ev: e });
    cur = e.endPos;
  }
  if (cur < txt.length) out.push({ text: txt.slice(cur), ev: null });
  return out;                               // 前端只做渲染，不重算任何特征
});
// 点击色块 → 高亮关联的 EvidenceCard（按 feature_code 归组）；
// 页面顶部固定展示"置信度 + 仅供参考声明"，避免教师只看分数不看证据。

9.3 检测详情页信息层级（强制顺序）

  1. 仅供参考声明条（置顶，不可折叠）：本结果由特征分析生成，仅供教师复核参考，不作为处理依据。
  2. 置信度 + 风险等级：置信度低时风险等级自动展示为「低关注 · 证据不足」。
  3. 特征雷达 + 特征明细表：每个特征给原始值、归一化分、权重、单条中性说明。
  4. 证据区：原文高亮 + 逐条证据卡（片段 / 何以成为证据 / 定位跳转）。
  5. 复核建议 + 可引用条文（来自 RAG，标注"条文引用仅供参考"）。
  6. 申诉入口：一键发起申诉工单（若该作业允许申诉）。

9.4 脱敏展示规范

  • 学号：前 6 位 + ******（如 2024 01******）
  • 姓名：保留姓 + *（如 张*）；详情页内亦不展示完整姓名
  • 下载导出：报告文件名用「课程_作业_学号后 4 位」，不含姓名
  • 列表页默认不展示原始分数，只展示关注等级与置信度
